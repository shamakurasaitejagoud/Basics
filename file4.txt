pipeline {
    agent any

    tools {
        maven 'MAVEN_HOME'
    }
    triggers {
        githubPush()
    }
    stages {
        stage('Checkout') {
            steps {
                git branch: 'main',
                    url: "https://github.com/saitejashamakura6002-hash/maven_web_project.git"
            }
        }

        stage('Build') {
            steps {
                dir('maven_web_project') {
                    bat 'mvn clean package'
                }
            }
        }

        stage('Test') {
            steps {
                dir('maven_web_project') {
                    bat 'mvn test'
                }
            }
        }

        stage('Docker Build') {
            steps {
                dir('maven_web_project') {
                    bat 'docker build -t webimage .'

                    bat 'docker tag webimage saiteja3156/webimage:v%BUILD_NUMBER%'
                    bat 'docker tag webimage saiteja3156/webimage:latest'
                }
            }
        }

        stage('Docker Login') {
            steps {
                withCredentials(
                    [usernamePassword(
                        credentialsId: 'dockerhub-credentials',
                        passwordVariable: 'DOCKER_PASSWORD',
                        usernameVariable: 'DOCKER_USERNAME'
                    )]
                ) {
                    bat 'docker login -u %DOCKER_USERNAME% -p %DOCKER_PASSWORD%'
                }
            }
        }

        stage('Docker Push') {
            steps {
                bat 'docker push saiteja3156/webimage:v%BUILD_NUMBER%'
                bat 'docker push saiteja3156/webimage:latest'
            }
        }

        stage('Docker Run') {
            steps {
                bat 'docker run -d -p 8094:8080 saiteja3156/webimage:latest'
            }
        }
    }

    post {
        success {
            mail to: 'shamakurasaiteja@gmail.com',
                 subject: "✅ Success: Pipeline ${env.JOB_NAME} [Build #${env.BUILD_NUMBER}]",
                 body: "The pipeline completed successfully!\n\nView the logs here: ${env.BUILD_URL}"
        }

        failure {
            mail to: 'shamakurasaiteja@gmail.com',
                 subject: "❌ Failure: Pipeline ${env.JOB_NAME} [Build #${env.BUILD_NUMBER}]",
                 body: "Attention team, the build has failed.\n\nCheck the console output directly to debug: ${env.BUILD_URL}console"
        }
    }
}
